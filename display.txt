# Apple II PC - 화면 설정. 고친 뒤 재부팅하면 적용됩니다.
#
# RESOLUTION: 화면(LCD 패널)의 원래 해상도
#   LCD는 원래 해상도로 켜야 모니터가 흐리게 늘리거나 줄이지 않습니다.
#   예) 800x600, 1024x768, 1280x1024, 1280x800, 1440x900, 1600x1200
#   auto 는 모니터가 알려주는 원래 해상도를 씁니다 (그래픽 카드가 지원할 때).
#   해상도가 높을수록 오래된 PC에서는 느려질 수 있습니다.
#   그래픽 카드에 그 해상도가 없으면 1024x768, 800x600 순서로 내려갑니다.
RESOLUTION=1024x768
#
# PRIORITY: 애플 화면을 키우는 방식
#   aspect  비율 우선. 실기 모니터처럼 4:3으로 화면을 가득 채웁니다.
#           배율이 정수가 아니면 줄 높이가 조금씩 다를 수 있습니다.
#   sharp   선명함 우선. 모든 줄 높이를 똑같이(세로 정수 배율) 하고
#           4:3을 지킵니다. 해상도에 따라 그림이 작아지고 테두리가 생깁니다.
PRIORITY=aspect
#
# 와이드 모니터가 화면을 옆으로 늘려 보여주면 실행 중에 Alt+F6 으로
# 모니터 모양(16:9, 16:10 ...)을 고르세요 (linapple.conf 의 Monitor Aspect).
